import json, time, urllib.request
from openai import OpenAI
import requests

# 1. ต่อยอดจาก Module 2: สั่ง LM Studio แตกไอเดียภาษาไทย 1 บรรทัดเป็น Storyboard 3 ฉาก (JSON)
llm = OpenAI(base_url="http://localhost:1234/v1", api_key="lm-studio")

DIRECTOR_SYSTEM_PROMPT = """You are an AI Film & Storyboard Director.
Given a Thai concept, return ONLY valid JSON with:
{
  "project_title_th": "...",
  "project_title_en": "...",
  "logline_th": "...",
  "color_palette": ["#hex1", "#hex2", "#hex3", "#hex4"],
  "scenes": [
    {
      "scene_number": 1,
      "title_th": "...",
      "camera_angle": "Wide Establishing Shot, 24mm lens",
      "narration_th": "...",
      "sd_positive_prompt": "detailed English prompt for Stable Diffusion 1.5...",
      "sd_negative_prompt": "blurry, watermark, text, deformed, low quality"
    }
  ]
} (Always generate exactly 3 scenes)."""

def direct_storyboard(thai_idea: str, style_keywords: str) -> dict:
    res = llm.chat.completions.create(
        model="local-model",
        messages=[
            {"role": "system", "content": DIRECTOR_SYSTEM_PROMPT},
            {"role": "user", "content": f"แนวคิด: {thai_idea}\nสไตล์ภาพ: {style_keywords}"}
        ],
        temperature=0.45,
        max_tokens=900
    )
    raw = res.choices[0].message.content.strip()
    return json.loads(raw[raw.find("{"):raw.rfind("}")+1])

# 2. ต่อยอดจาก Module 3 & 4: วนลูปส่งแต่ละฉากเข้า ComfyUI API (512x512, RTX 3050 6GB Safe)
def render_scene_comfyui(workflow_template: dict, scene: dict, seed: int) -> bytes:
    wf = json.loads(json.dumps(workflow_template))
    wf["6"]["inputs"]["text"] = scene["sd_positive_prompt"]
    wf["7"]["inputs"]["text"] = scene["sd_negative_prompt"]
    wf["3"]["inputs"]["seed"] = seed
    wf["3"]["inputs"]["steps"] = 20
    wf["3"]["inputs"]["cfg"] = 7.5

    r = requests.post("http://127.0.0.1:8188/prompt", json={"prompt": wf}, timeout=10)
    prompt_id = r.json()["prompt_id"]
    while True:
        hist = requests.get(f"http://127.0.0.1:8188/history/{prompt_id}", timeout=5).json()
        if prompt_id in hist:
            img_info = hist[prompt_id]["outputs"]["9"]["images"][0]
            view_url = f"http://127.0.0.1:8188/view?filename={img_info['filename']}&subfolder={img_info['subfolder']}&type={img_info['type']}"
            return urllib.request.urlopen(view_url).read()
        time.sleep(1.0)
Self-Study Blueprint

วิธีนำโค้ดจากคลาส 6 ชั่วโมง มาประกอบเป็นโปรเจกต์นี้

CineLocal Studio ใช้เฉพาะไลบรารีมาตรฐานที่อยู่ในไฟล์ requirements.txt ของคลาสเรียน โดยนำโค้ดจากทั้ง 4 โมดูลมาต่อกันตามลำดับนี้:

ขั้นที่ 1: ปรับ System Prompt ให้ส่งกลับเป็นลิสต์ 3 ฉาก (ต่อยอด Module 2)

ในห้องเรียนเราสั่งให้ LM Studio ตอบกลับเป็น JSON ที่มี positive_prompt เดียว ส่วนในโปรเจกต์นี้เราแก้ SYSTEM_PROMPT ให้ส่งลิสต์ "scenes": [...] ออกมา 3 ฉาก พร้อมระบุมุมกล้อง (camera_angle) คำบรรยายฉากภาษาไทย (narration_th) และชุดสีประจำเรื่อง (color_palette)

ขั้นที่ 2: ใช้ For-Loop ส่งเรนเดอร์ทีละฉากเพื่อไม่ให้ VRAM 6GB ล้น (ต่อยอด Module 3 & 4)

ถ้าตั้งค่า batch_size = 3 ในโหนด Empty Latent Image การ์ดจอ RTX 3050 (6GB VRAM) มีโอกาสค้างหรือเกิด Out of Memory เราจึงใช้ลูป for scene in storyboard["scenes"]: ใน Python ยิง HTTP POST ไปที่ /prompt ทีละฉากแทน และขยับค่า base_seed + index เพื่อให้ภาพทั้ง 3 ฉากคุมโทนใกล้เคียงกันโดยใช้ VRAM ประมาณ 4.2 GB

ขั้นที่ 3: รันสคริปต์จริงจากโฟลเดอร์ 05-bonus-showcase/

ในโปรเจกต์มีไฟล์ 05-bonus-showcase/app.py เตรียมไว้ให้แล้ว เปิด LM Studio (พอร์ต 1234) กับ ComfyUI (พอร์ต 8188) แล้วรันคำสั่งด้านล่างได้เลย: 

python 05-bonus-showcase/app.py
